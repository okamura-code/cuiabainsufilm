# v5 "Cinema e peso" Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Hero em Three.js dirigido pelo scroll, intro termômetro, raio-X do vidro em 3 atos, termômetro lateral, antes/depois automático e line-up pinado dos níveis.

**Architecture:** Astro 7 estático; cada cena é um componente `.astro` com script inline `data-astro-rerun` (padrão do repo) lendo um progresso `p` 0–1 da seção pinada (`sticky` dentro de seção alta). Three.js é a única dependência nova, importada dinamicamente após `load` em `src/scripts/hero3d.ts`. Nada de GSAP.

**Tech Stack:** Astro 7, TypeScript, Three.js (npm `three`, GLTFLoader + DRACOLoader), CSS scroll-driven onde já existe, Lenis já presente.

```ts
/* Hero 3D: SUV CC0 iluminado por um "sol" que vai do zênite ao horizonte com o scroll; o vidro escurece junto.
   Carrega Three.js só depois do load. Sem WebGL ou com reduced-motion a foto de poster fica. */
const PIN = '(min-width: 900px) and (min-height: 640px) and (prefers-reduced-motion: no-preference)';
const MOBILE = '(max-width: 899px) and (prefers-reduced-motion: no-preference)';
let started = false;
async function start() {
  const hero = document.getElementById('topo');
  const cv = hero?.querySelector<HTMLCanvasElement>('.hero__3d');
  if (started || !hero || !cv) return;
  const desktop = matchMedia(PIN).matches, mobile = matchMedia(MOBILE).matches;
  if (!desktop && !mobile) return;
  const gl = document.createElement('canvas').getContext('webgl2'); if (!gl) return;
  started = true;
  const THREE = await import('three');
  const { GLTFLoader } = await import('three/examples/jsm/loaders/GLTFLoader.js');
  const { DRACOLoader } = await import('three/examples/jsm/loaders/DRACOLoader.js');
  const renderer = new THREE.WebGLRenderer({ canvas: cv, antialias: desktop, alpha: true, powerPreference: 'low-power' });
  renderer.setPixelRatio(desktop ? Math.min(devicePixelRatio, 2) : 1);
  renderer.shadowMap.enabled = desktop;
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  const scene = new THREE.Scene();
  const cam = new THREE.PerspectiveCamera(32, 1, 0.1, 100);
  const sun = new THREE.DirectionalLight(0xffffff, 3); sun.castShadow = desktop; scene.add(sun);
  scene.add(new THREE.HemisphereLight(0xb4aea2, 0x0b0d10, 0.6));
  const ground = new THREE.Mesh(new THREE.CircleGeometry(12, 48), new THREE.MeshStandardMaterial({ color: 0x15181d, roughness: 0.35, metalness: 0.2 }));
  ground.rotation.x = -Math.PI / 2; ground.receiveShadow = desktop; scene.add(ground);
  const glass: import('three').MeshPhysicalMaterial[] = [];
  const draco = new DRACOLoader(); draco.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.7/');
  const loader = new GLTFLoader(); loader.setDRACOLoader(draco);
  const gltf = await loader.loadAsync('/3d/car.glb');
  const car = gltf.scene; car.traverse((o) => {
    const m = o as import('three').Mesh; if (!m.isMesh) return;
    m.castShadow = desktop;
    if (/glass|window/i.test(m.name)) { const g = new THREE.MeshPhysicalMaterial({ color: 0xffffff, transmission: 0.9, roughness: 0.05, thickness: 0.2, transparent: true, opacity: 0.35 }); m.material = g; glass.push(g); }
  });
  const box = new THREE.Box3().setFromObject(car); const size = box.getSize(new THREE.Vector3()); const s = 4.2 / size.x; car.scale.setScalar(s);
  car.position.y = -box.min.y * s; scene.add(car);
  let yaw = 0, mx = 0, my = 0;
  const p = () => { const span = hero.offsetHeight - innerHeight; return span > 0 ? Math.min(1, Math.max(0, -hero.getBoundingClientRect().top / span)) : 0; };
  function resize() { const w = cv.clientWidth, h = cv.clientHeight; renderer.setSize(w, h, false); cam.aspect = w / h; cam.updateProjectionMatrix(); }
  addEventListener('resize', resize); resize();
  if (desktop) addEventListener('pointermove', (e) => { mx = e.clientX / innerWidth - 0.5; my = e.clientY / innerHeight - 0.5; }, { passive: true });
  else { let x0 = 0, y0 = yaw; cv.addEventListener('pointerdown', (e) => { x0 = e.clientX; y0 = yaw; }); cv.addEventListener('pointermove', (e) => { if (e.buttons) yaw = y0 + (e.clientX - x0) / 150; }); }
  const warm = new THREE.Color(0xff9a3c), white = new THREE.Color(0xffffff), dark = new THREE.Color(0x0b0d10);
  function frame() {
    const t = p(); (window as any).__heroP = t;
    const a = Math.PI / 2 - t * 1.25; sun.position.set(Math.cos(a) * 8, Math.sin(a) * 8 + 0.5, 4); sun.color.copy(white).lerp(warm, t); sun.intensity = 3 - t * 1.6;
    glass.forEach((g) => { g.color.copy(white).lerp(dark, t); g.opacity = 0.35 + t * 0.55; g.transmission = 0.9 - t * 0.75; });
    car.rotation.y = -0.6 + t * 0.6 + yaw;
    cam.position.set(6 + mx * 0.8, 1.6 + my * 0.4, 6); cam.lookAt(0, 0.6, 0);
    renderer.render(scene, cam); requestAnimationFrame(frame);
  }
  frame(); cv.classList.add('is-on');
}
if (document.readyState === 'complete') start(); else addEventListener('load', start, { once: true });
document.addEventListener('astro:page-load', () => { started = false; start(); });
```
